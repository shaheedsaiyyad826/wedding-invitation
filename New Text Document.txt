<script>

let currentPage = 1;

function nextPage(page) {
  document.getElementById('page' + currentPage).classList.remove('active');
  document.getElementById('music' + currentPage).pause();

  currentPage = page;

  document.getElementById('page' + currentPage).classList.add('active');
  document.getElementById('music' + currentPage).play();
}

function openMap() {
  window.open("https://www.google.com/maps/search/?api=1&query=Roshani+Palace+Kudachi");
}

// Countdown Timer
let weddingDate = new Date("2026-05-30T12:00:00").getTime();

let timer = setInterval(function() {
  let now = new Date().getTime();
  let distance = weddingDate - now;

  let days = Math.floor(distance / (1000 * 60 * 60 * 24));
  let hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  let minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
  let seconds = Math.floor((distance % (1000 * 60)) / 1000);

  document.getElementById("countdown").innerHTML =
    days + "d " + hours + "h " + minutes + "m " + seconds + "s ";

  if (distance < 0) {
    clearInterval(timer);
    document.getElementById("countdown").innerHTML = "Wedding Started!";
  }
}, 1000);

// Auto play first music
window.onload = () => {
  document.getElementById('music1').play();
};
